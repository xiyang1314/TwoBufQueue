1// WarmPool：一个对象资源池。2// 池里的对象被重复使用，避免高频 new/delete；借出时对象先被 reset() 复位成"干净"状态，3// 用完归还，对象回到池里等待下次借出。适合消息对象、连接对象这类反复借还的场景。45#include "TwoBufQueue/src/base_components/WarmPool/warm_pool.hpp"6#include <cstdint>7#include <cstdio>8#include <stdexcept>9using namespace TwoBufQueue;1011// 放进池里的对象只需要实现 reset(generation)：每次借出前调用，把对象恢复成初始状态。12// 返回 false 表示这次借出失败（对象会被安排延迟销毁，本次拿不到）。13struct Message {14    int value = 0;15    bool reset(std::uint64_t) noexcept { value = 0; return true; }16};1718int main() try {19    //--------------- 1. 初始化：至少 2 个、最多 8 个对象 -----------------------20    // 构造时就会立即预分配 min_count(2) 个对象，避免运行时才慢慢创建。21    WarmPool<Message> pool(2, 8);2223    //--------------- 2. 借出：get() 返回一个 Loan（借条），不拥有对象 ---------24    auto loan = pool.get();25    if (!loan) throw std::runtime_error("get failed");   // 池满或创建失败时为空26    loan->value = 42;                                    // 像指针一样访问对象27    if (!pool.is_valid(loan)) throw std::runtime_error("loan should be valid");2829    //--------------- 3. 归还：finish()（等价 recycle()）-----------------------30    // finish 是 [[nodiscard]]，必须检查返回值：true 表示归还成功。31    if (!loan.finish()) throw std::runtime_error("recycle failed");3233    // 再次借出：对象会复用上一次归还的那个，reset() 已经把 value 清成 0。34    auto again = pool.get();35    if (!again || again->value != 0) throw std::runtime_error("reset not applied on reuse");36    if (!again.finish()) throw std::runtime_error("second recycle failed");3738    //--------------- 4. 查询与预分配 ------------------------------------------39    if (!pool.all_released()) throw std::runtime_error("not all released");40    // 提前把在池对象数补齐到 4 个，减少后续 get() 的动态分配开销。41    if (!pool.preallocate(4)) throw std::runtime_error("preallocate failed");42    if (pool.resource_count() != 4) throw std::runtime_error("resource_count mismatch");4344    std::puts("PASS: borrow, recycle, reuse and preallocate");45} catch (const std::exception& error) {46    std::fprintf(stderr, "%s\n", error.what());47    return 1;48}
