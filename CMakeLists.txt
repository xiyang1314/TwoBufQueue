include(GNUInstallDirs)
find_package(Threads REQUIRED)
file(GLOB_RECURSE component_sources CONFIGURE_DEPENDS "src/*.cpp")
add_library(TwoBufQueue STATIC ${component_sources})
add_library(TwoBufQueue::TwoBufQueue ALIAS TwoBufQueue)
target_compile_features(TwoBufQueue PUBLIC cxx_std_20)
set_target_properties(TwoBufQueue PROPERTIES CXX_EXTENSIONS OFF)
# Forwarders preserve existing include paths without requiring a particular checkout
# directory name or Windows symlink privileges. The original headers remain authoritative.
file(GLOB_RECURSE component_headers CONFIGURE_DEPENDS "src/*.hpp")
foreach(header IN LISTS component_headers)
  file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/TwoBufQueue/${relative}"
    CONTENT "#pragma once\n#include \"${header}\"\n")
endforeach()
target_include_directories(TwoBufQueue PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_link_libraries(TwoBufQueue PUBLIC Threads::Threads)
if(BRQ_ENABLE_POOL_DEPTH_STATS)
  # Keep all translation units consistent; default library builds omit the
  # fields, branch and counters entirely.
  target_compile_definitions(TwoBufQueue PUBLIC BRQ_ENABLE_POOL_DEPTH_STATS=1)
endif()
if(WIN32)
  target_link_libraries(TwoBufQueue PUBLIC synchronization)
endif()
if(BRQ_DISABLE_TSC)
  target_compile_definitions(TwoBufQueue PRIVATE FAST_LOG_DISABLE_TSC)
endif()

function(brq_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endfunction()
